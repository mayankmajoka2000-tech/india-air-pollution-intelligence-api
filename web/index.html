Quick start

No key needed. Call it from anywhere.

cURL
# 10 worst stations in Delhi right now
curl "BASE/live/stations?state=Delhi&limit=10"

# Official CPCB AQI from concentrations
curl -X POST "BASE/aqi/calculate" \
  -H "Content-Type: application/json" \
  -d '{"pm25": 92, "pm10": 180, "no2": 45}'
Python
import requests

BASE = "BASE"
cities = requests.get(f"{BASE}/live/cities",
                      params={"limit": 5}).json()

for c in cities["cities"]:
    print(c["city"], c["mean_aqi"], c["category"])
